FUNCTION UserManager.load_users():
    TRY:
        OPEN 'users.txt' AS file
        FOR EACH line IN file:
            username, password = SPLIT line BY '|'
            users[username] = password
    EXCEPTION FileNotFoundError:
        PASS
    RETURN users

```plaintext
FUNCTION register():
    IF request.method IS POST THEN
        username = GET form data 'username'
        password = GET form data 'password'
        IF user_manager.register(username, password) THEN
            REDIRECT to login
        ELSE
            RETURN "Username already exists!"

FUNCTION UserManager.register(username, password):
    IF username IN users THEN
        RETURN False
    users[username] = password
    CALL save_users()
    RETURN True

FUNCTION UserManager.save_users():
    OPEN 'users.txt' AS file
    FOR EACH username, password IN users.items():
        WRITE f"{username}|{password}\n" TO file

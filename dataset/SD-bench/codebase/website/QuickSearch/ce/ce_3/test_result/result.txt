    ```
        username = request.form['username']
        password = request.form['password']
        users = load_users()

        IF username IS IN users AND users[username] == password THEN
          session['username'] = username
          REDIRECT TO search_books
        END IF
      END IF
      RENDER login.html
    END FUNCTION

    FUNCTION load_users()
      users = {}
      OPEN 'users.txt' AS file
        FOR line IN file DO
          username, password = line.strip().split('|')
          users[username] = password
        END FOR
      RETURN users
    END FUNCTION
    ```

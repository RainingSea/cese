```pseudocode
CLASS BookManager
    FUNCTION add_book(title: str, author: str, year: int) -> bool
        IF title already exists in self.books
            RETURN False
        ENDIF
        APPEND (title, author, year) to self.books
        OPEN 'books.txt' in append mode
            WRITE title|author|year to file
        CLOSE file
        RETURN True
    END FUNCTION

    FUNCTION get_books() -> list
        RETURN self.books
    END FUNCTION

ROUTE /book_management
    IF request.method == 'POST'
        title = request.form['title']
        author = request.form['author']
        year = request.form['year']
        IF book_manager.add_book(title, author, int(year))
            FLASH 'Book added successfully.'
        ELSE
            FLASH 'Failed to add book.'
        ENDIF
    ENDIF
    books = book_manager.get_books()
    RENDER book_management.html with books
END ROUTE
```

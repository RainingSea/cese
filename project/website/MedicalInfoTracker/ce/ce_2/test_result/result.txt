    ```
    CLASS MedicalInfoManager
        FUNCTION __init__()
            LOAD medical info from 'medical_info.txt'

        FUNCTION load_medical_info()
            OPEN 'medical_info.txt'
            FOR EACH line IN file
                SPLIT line into diagnosis, medication, treatment
                APPEND [diagnosis, medication, treatment] to medical_info list

        FUNCTION add_info(diagnosis, medication, treatment)
            APPEND [diagnosis, medication, treatment] to medical_info list
            APPEND diagnosis|medication|treatment to 'medical_info.txt'

        FUNCTION view_info()
            RETURN medical_info list

    @app.route('/medical_info', methods=['GET', 'POST'])
    FUNCTION medical_info()
        IF request method is POST
            CALL add_info with inputs from form
        info = CALL view_info
        RENDER 'medical_info.html' with info
    ```

- **test_set_reminders**
  - **Project Code Pseudocode**:
    ```
    CLASS ReminderManager
        FUNCTION __init__()
            LOAD reminders from 'reminders.txt'

        FUNCTION load_reminders()
            OPEN 'reminders.txt'
            FOR EACH line IN file
                SPLIT line into date, time, description
                APPEND [date, time, description] to reminders list

        FUNCTION set_reminder(date, time, description)
            APPEND [date, time, description] to reminders list
            APPEND date|time|description to 'reminders.txt'

        FUNCTION get_reminders()
            RETURN reminders list

    @app.route('/reminders', methods=['GET', 'POST'])
    FUNCTION reminders()
        IF request method is POST
            CALL set_reminder with inputs from form
        reminders = CALL get_reminders
        RENDER 'reminders.html' with reminders
    ```

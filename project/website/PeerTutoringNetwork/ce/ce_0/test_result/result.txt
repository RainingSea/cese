FUNCTION register()
    IF METHOD IS POST THEN
        username = request.form['username']
        password = request.form['password']
        email = request.form['email']
        
        IF user_manager.register(username, password, email) THEN
            RETURN redirect(url_for('login'))
        ENDIF
    ENDIF
    RENDER registration.html
END FUNCTION

CLASS UserManager
    FUNCTION register(username: str, password: str, email: str) -> bool
        IF username already exists in users THEN
            RETURN False
        ENDIF
        new_user = User(username, password, email)
        APPEND new_user to users list
        CALL save_users()
        RETURN True
    END FUNCTION

    FUNCTION save_users()
        OPEN 'users.txt' FOR WRITING
            FOR EACH user in users
                WRITE user.username, user.password, user.email TO file
            END FOR
        CLOSE file
    END FUNCTION
END CLASS

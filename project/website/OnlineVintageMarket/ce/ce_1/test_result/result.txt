CLASS UserManager:

    FUNCTION load_users() -> list:
        INITIALIZE users AS empty list
        TRY:
            OPEN 'users.txt' in read mode AS file:
                FOR each line IN file:
                    SPLIT line by '|'
                    APPEND User(username, password) TO users
        EXCEPT FileNotFoundError:
            PASS
        RETURN users

    FUNCTION add_user(username: str, password: str):

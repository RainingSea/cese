```pseudocode
FUNCTION main()
    INITIALIZE Flask application
    SET secret_key for session management

    FUNCTION load_users()
        INITIALIZE empty dictionary users
        OPEN 'users.txt' for reading
            FOR each line in file
                SPLIT line by '|'
                ASSIGN username and password 
                ADD username and password to users
        RETURN users

    FUNCTION load_books()
        INITIALIZE empty list books
        OPEN 'books.txt' for reading
            FOR each line in file
                SPLIT line by '|'
                ASSIGN title, author, summary
                CREATE Book object with(title, author, summary)
                APPEND Book object to books
        RETURN books

    users_data = load_users()
    books_data = load_books()

    DEFINE route for '/'
        RETURN rendered 'login.html'

    DEFINE route for '/dashboard' with methods ['GET', 'POST']
        IF request.method is 'POST'
            GET username and password from form
            IF username exists in users_data AND password matches
                SET session['username'] to username
                RETURN rendered 'dashboard.html' with books_data
        REDIRECT to '/'

    START Flask app
END FUNCTION
```

#### User Registration and Login Management

**Files involved:** `main.py` (UserManager class), `register.html`, `login.html`

**Pseudocode:**

```
CLASS UserManager
    FUNCTION register(username, password)
        IF user_exists(username)
            RETURN False
        ENDIF
        user = { "username": username, "password": password }
        append user as JSON line to USERS_FILE
        RETURN True
    END

    FUNCTION login(username, password)
        users = load all users from USERS_FILE (each line JSON decoded)
        FOR each user in users
            IF user.username == username AND user.password == password
                RETURN True
            ENDIF
        ENDFOR
        RETURN False
    END

    FUNCTION user_exists(username)
        users = load all users from USERS_FILE
        FOR each user in users
            IF user.username == username
                RETURN True
            ENDIF
        ENDFOR
        RETURN False
    END

    FUNCTION _load_users()
        IF USERS_FILE does not exist
            RETURN empty list
        ENDIF
        OPEN USERS_FILE for reading
        FOR each line in file
            TRY parse JSON line as user
            ADD user to users list
            EXCEPT JSON error: skip line
        ENDFOR
        RETURN users list
    END

    FUNCTION _save_user(user)
        OPEN USERS_FILE for appending
        WRITE JSON string of user + newline
    END
ENDCLASS
```

```
CLASS BookManager
    FUNCTION search_books(query)
        query_lower = lowercase(query)
        books = load all books from BOOKS_FILE (each line JSON decoded)
```

```
CLASS ReadingListManager
    FUNCTION add_book(username, book_id)
        entries = load reading list entries from READING_LIST_FILE
        FOR each entry in entries
            IF entry.username == username AND entry.book_id == book_id
                RETURN False  # already in list
            ENDIF
        ENDFOR
        append {username, book_id} as JSON line to READING_LIST_FILE
        RETURN True
    END

    FUNCTION get_reading_list(username)
        entries = load reading list entries
        book_ids = []
        FOR each entry in entries
            IF entry.username == username
                ADD entry.book_id to book_ids
            ENDIF
        ENDFOR
        RETURN book_ids
    END

    FUNCTION remove_book(username, book_id)
        entries = load reading list entries
        new_entries = entries excluding those matching username and book_id
        IF length(new_entries) == length(entries)
            RETURN False  # no removal
        ENDIF
        overwrite READING_LIST_FILE with new_entries as JSON lines
        RETURN True
    END

    FUNCTION _load_reading_list()
        IF READING_LIST_FILE does not exist
            RETURN empty list
        ENDIF
        OPEN READING_LIST_FILE for reading
        FOR each line
            TRY parse JSON line as entry
            ADD entry to entries list
            EXCEPT JSON error: skip line
        ENDFOR
        RETURN entries list
    END

    FUNCTION _save_reading_list_entry(entry)
        OPEN READING_LIST_FILE for appending
        WRITE JSON string of entry + newline
    END

    FUNCTION _save_all_reading_list(entries)
        OPEN READING_LIST_FILE for writing (overwrite)
        FOR each entry in entries
            WRITE JSON string of entry + newline
        ENDFOR
    END
ENDCLASS
```

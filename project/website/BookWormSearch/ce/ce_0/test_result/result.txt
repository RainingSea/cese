```plaintext
CLASS UserManager
    FUNCTION __init__(users_file)
        SET self.users_file = users_file
        SET self.users = load_users()

    FUNCTION login(username, password)
        RETURN self.users.get(username) == password

    FUNCTION load_users()
        SET users = {}
        TRY
            OPEN self.users_file AS f
                FOR EACH line IN f
                    SPLIT line BY '|'
                    SET users[username] = password
        EXCEPT FileNotFoundError
            PASS
        RETURN users

CLASS Main
    FUNCTION setup_routes()
        @self.app.route('/login', methods=['POST'])
        FUNCTION do_login()
            SET username = request.form['username']
            SET password = request.form['password']
            IF self.user_manager.login(username, password)
                SET session['username'] = username
                RETURN redirect('/dashboard')
            RETURN redirect('/')
```

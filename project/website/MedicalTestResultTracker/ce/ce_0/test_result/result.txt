```plaintext
FUNCTION register(username: STRING, password: STRING) -> BOOLEAN
    INITIALIZE user_manager as UserManager
    IF request method is POST
        SET username to request.username
        SET password to request.password

        IF user_manager.register(username, password) IS TRUE
            REDIRECT to login page
        ELSE
            RETURN "Username already exists"
        ENDIF
    ENDIF
    RENDER register.html template
END FUNCTION

CLASS UserManager
    FUNCTION __init__()
        SET users to load_users()

    FUNCTION load_users() -> LIST
        IF 'users.txt' does not exist
            RETURN empty LIST
        OPEN 'users.txt' for reading AS file
            RETURN each line stripped and split by '|'

    FUNCTION register(username: STRING, password: STRING) -> BOOLEAN
        FOR each user in users
            IF user[0] == username
                RETURN FALSE  // User already exists
            ENDFOR
        APPEND [username, password] to users
        OPEN 'users.txt' for appending AS file
            WRITE "{username}|{password}\n" to file
        RETURN TRUE  // Successful registration
END CLASS
```

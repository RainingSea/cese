class UserManager:
    initialize(users_file: string):
        self.users_file = users_file
        self.users = load_users()

    load_users() -> dict:
        users = empty dictionary
        try:
            open users_file for reading:
                for each line in file:
                    split line into username, password, email, applied_jobs
                    create User(username, password, email)
                    initialize user.applied_jobs with applied_jobs if exists
                    add user to users dictionary with username as key
        catch FileNotFoundError:
            pass
        return users

    register_user(username: string, password: string, email: string) -> bool:
        if username exists in users:
            return False
        create new User with username, password, email
        add new_user to users dictionary
        open users_file for appending:
            write new_user.to_string() to file
        return True

```python
FUNCTION UserManager()
    FILE: users.txt
    IF NOT EXISTS users.txt
        RETURN EMPTY LIST
    
    OPEN users.txt AS file
        FOR EACH LINE IN file
            SPLIT line BY '|' INTO user_data
            APPEND user_data TO users
    END FOR

FUNCTION register(username, password):
    FOR EACH user IN users
        IF user[0] == username
            RETURN False // User exists

    APPEND [username, password] TO users
    OPEN users.txt IN APPEND mode AS file
        WRITE username + "|" + password + "\n"
    RETURN True

FUNCTION register_route()
    IF request.method == 'POST'
        username = request.form['username']
        password = request.form['password']
        IF UserManager.register(username, password)
            RETURN redirect('/')

RENDER registration.html
```

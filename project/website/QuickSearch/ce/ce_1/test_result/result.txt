            IF book_title is not in self.reading_lists[username]
                APPEND book_title to self.reading_lists[username]
        ELSE

        CALL save_reading_lists()
        RETURN TRUE

    FUNCTION get_reading_list(username: str) -> list

    FUNCTION load_reading_lists() -> dict

        TRY
            OPEN file with self.filename as file
                FOR each line in file
                    SPLIT line by '|' into username and books
                    SET reading_lists[username] = SPLIT books by ','
        EXCEPT FileNotFoundError
            PASS
        RETURN reading_lists

    FUNCTION save_reading_lists()
        OPEN file with self.filename as file
            FOR each username, books in self.reading_lists.items()

```python
class User:
    users_file = 'users.txt'

    # Constructor
    FUNCTION User(username: str, password: str)
        SET self.username = username
        SET self.password = password
    END FUNCTION

    # Load users from a file
    FUNCTION load_users() RETURNS users_dict
        SET users_dict = {}
        IF os.path.exists(users_file)
            OPEN users_file IN READ MODE AS file
                FOR line IN file
                    SPLIT line BY '|'
                    SET username = VALUE[0]
                    SET password = VALUE[1]
                    SET users_dict[username] = password
                END FOR
            END OPEN file
        RETURN users_dict
    END FUNCTION

    # Register a new user
    FUNCTION register() RETURNS BOOL
        IF self.username NOT IN load_users()
            OPEN users_file IN APPEND MODE AS file
                WRITE self.username + '|' + self.password + '\n' TO file
            END OPEN file
            RETURN True
        END IF
        RETURN False
    END FUNCTION
```

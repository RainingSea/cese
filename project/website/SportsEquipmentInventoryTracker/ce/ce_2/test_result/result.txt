        RETURN equipment list

        APPEND {name, quantity, condition, location} to equipment list
        OPEN file 'equipment.txt' in append mode

        CLOSE file
        RETURN True
    ENDFUNCTION

        RETURN equipment list
    ENDFUNCTION
ENDCLASS

        PARSE users into (username, password) tuples
        STORE users in self.users

        FOR each (user, pwd) IN self.users
            IF user == username AND pwd == password
                RETURN True
            ENDIF
        ENDFOR
        RETURN False
    END FUNCTION 

CLASS ContentManager
    FUNCTION __init__()
        LOAD tips from 'tips.txt'
        LOAD articles from 'articles.txt'
        LOAD forum posts from 'forum.txt'
    END FUNCTION

    FUNCTION load_users()
        OPEN 'users.txt'
        READ file lines
        FOR each line IN lines
            SPLIT line by '|'
            APPEND (username, password) to users list
        ENDFOR
        RETURN users
    END FUNCTION

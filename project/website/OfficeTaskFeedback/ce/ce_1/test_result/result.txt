                    SPLIT line BY '|'

            END FOR
        EXCEPTION FileNotFoundError
            PASS
        RETURN feedbacks

    FUNCTION review_feedback() -> LIST

        GET category FROM request.form
        GET content FROM request.form

    ENDIF
    RENDER feedback.html

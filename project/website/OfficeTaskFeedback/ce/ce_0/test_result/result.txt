                    APPEND (feedback_id, user_id, content, category) TO feedbacks
        EXCEPT FileNotFoundError
            PASS
        RETURN feedbacks

    FUNCTION review_feedback() -> LIST
        RETURN feedbacks

CLASS UserManager
    FUNCTION __init__()
        INITIALIZE users
        CALL load_users()

    FUNCTION load_users()
        INITIALIZE users AS EMPTY LIST
        TRY
            OPEN 'users.txt' FOR READING AS file
                FOR EACH line IN file
                    SPLIT line BY '|'
                    APPEND (username, password) TO users
        EXCEPT FileNotFoundError
            PASS
        RETURN users

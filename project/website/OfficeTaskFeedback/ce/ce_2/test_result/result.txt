CLASS FeedbackManager
    FUNCTION __init__()
        self.feedbacks = self.load_feedbacks()

    FUNCTION load_feedbacks()
        DECLARE feedbacks AS LIST
        TRY
            OPEN 'feedback.txt' AS file
                FOR EACH line IN file
                    APPEND line.strip() TO feedbacks
            END FOR
        EXCEPT FileNotFoundError
            PASS
        RETURN feedbacks
    END FUNCTION

    FUNCTION submit_feedback(username: STRING, feedback: STRING, category: STRING) -> BOOL
        CREATE feedback_entry AS STRING = username + "|" + feedback + "|" + category
        APPEND feedback_entry TO self.feedbacks
        OPEN 'feedback.txt' AS file with mode 'a'
            WRITE feedback_entry + "\n" TO file
        RETURN TRUE
    END FUNCTION

    FUNCTION get_feedbacks()
        RETURN self.feedbacks
    END FUNCTION
END CLASS

CLASS MainApp

    FUNCTION feedback_submission()
        IF request.method IS 'POST'
            username = session.get('username')
            feedback = request.form['feedback']
            category = request.form['category']
            feedback_manager.submit_feedback(username, feedback, category)
            REDIRECT to feedback_review
        END IF
        RENDER feedback_submission.html
    END FUNCTION

    FUNCTION feedback_review()
        feedbacks = feedback_manager.get_feedbacks()
        RENDER feedback_review.html WITH feedbacks
    END FUNCTION

END CLASS

CLASS UserManager
    FUNCTION __init__()
        CALL load_users()

    FUNCTION load_users()
        OPEN 'users.txt' AS file
            FOR each line IN file
                SPLIT line BY '|' INTO username, password
                APPEND {'username': username, 'password': password} TO users
        END OPEN
        RETURN users

    FUNCTION login(username, password) -> bool
        FOR each user IN users
            IF user['username'] EQUALS username AND user['password'] EQUALS password
                RETURN True
        RETURN False

CLASS FreelancerManager
    FUNCTION __init__()
        CALL load_freelancers()

    FUNCTION load_freelancers()
        OPEN 'freelancers.txt' AS file
            FOR each line IN file
                SPLIT line BY '|' INTO id, name, skills
                APPEND {'id': INTEGER(id), 'name': name, 'skills': SPLIT(skills BY ',')} TO freelancers
        END OPEN
        RETURN freelancers

APP Flask app
    SET secret_key TO 'your_secret_key'

    @APP.route('/', methods=['GET', 'POST'])
    FUNCTION login()
        IF request.method EQUALS 'POST'
            username = request.form['username']
            password = request.form['password']
            IF user_manager.login(username, password)
                SET session['username'] TO username
                RETURN redirect(url_for('home'))
        RETURN render_template('login.html')

    @APP.route('/home')
    FUNCTION home()
        RETURN render_template('home.html')

IF __name__ EQUALS '__main__':
    CALL app.run(port=8168, debug=False)

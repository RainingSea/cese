    FUNCTION load_users()
        OPEN 'users.txt' IN read mode AS file

                username, password = line.strip().split('|')
                self.users[username] = password
            END FOR
        END
    END FUNCTION
END CLASS

CLASS MovieManager
    FUNCTION load_movies()
        OPEN 'movies.txt' IN read mode AS file
            FOR line IN file
                title, description, rating = line.strip().split('|')
                self.movies[title] = {'description': description, 'rating': rating}
            END FOR
        END
    END FUNCTION

    FUNCTION load_favorites()
        OPEN 'favorites.txt' IN read mode AS file
            FOR line IN file
                username, movie_title = line.strip().split('|')
                IF username NOT IN self.favorites THEN
                    self.favorites[username] = []
                ENDIF
                APPEND movie_title TO self.favorites[username]
            END FOR
        END
    END FUNCTION
END CLASS

```plaintext
FUNCTION register(username: STRING, password: STRING) RETURNS BOOLEAN
    IF username NOT IN users THEN
        users[username] = password
        APPEND username|password TO users.txt
        RETURN True
    ELSE
        RETURN False

FUNCTION main()
    IF request.method IS POST THEN
        username = request.form['username']
        password = request.form['password']
        IF register(username, password) IS True THEN
            REDIRECT to login page
        ELSE
            OUTPUT 'Username already exists'
    RETURN render_template('register.html')
```

```plaintext
FUNCTION get_recommendations(preferences: DICT) RETURNS LIST
    recommendations = []
    FOR EACH movie IN movies DO
        IF movie.genre IS IN preferences['genres'] THEN
            APPEND movie TO recommendations
    RETURN recommendations

FUNCTION main()
```

```
FUNCTION search_movies(query)
    READ movies.txt
    FOR EACH movie IN movies
        IF query.lower() IN movie.title.lower()
            ADD movie TO results
    RETURN results

ROUTE /search (GET/POST)
    IF NOT logged_in: REDIRECT /login
    IF POST:
        query = request.form['query']
        results = search_movies(query)
    RENDER search.html WITH results
```

**test_07_view_favorites_list (ok)**
```
FUNCTION read_favorites(username)
    READ favorites.txt
    FOR EACH line IN file
        IF line.user == username
            ADD title TO favorites
    RETURN favorites

FUNCTION get_favorites()
    IF NOT current_user: RETURN []
    favorites = read_favorites(current_user)
    movies = read_movies()
    RETURN [movie IN movies IF movie.title IN favorites]

ROUTE /favorites (GET)
    IF NOT logged_in: REDIRECT /login
    favorites = get_favorites()
    RENDER favorites.html WITH favorites
```

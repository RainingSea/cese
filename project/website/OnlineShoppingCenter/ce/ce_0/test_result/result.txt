# UserManager.py
CLASS UserManager:
    FUNCTION validate_user(username, password):
        TRY:
            OPEN users.txt
            FOR EACH line IN file:
                SPLIT line INTO [stored_user, stored_pass, email]
                IF stored_user == username AND stored_pass == password:
                    RETURN True
        EXCEPT FileNotFoundError:
            RETURN False
        RETURN False

# main.py
@app.route('/login', methods=['POST'])
FUNCTION login():
    IF request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        IF user_manager.validate_user(username, password):
            SET session['username'] = username
            REDIRECT TO products
        ELSE:
            RENDER login.html WITH error="Invalid credentials"
    ELSE:
        RENDER login.html
```

# CartManager.py
CLASS CartManager:
    FUNCTION get_cart(user):
        carts = _read_carts()
        RETURN carts.get(user, [])

    FUNCTION _read_carts():
        carts = {}
        TRY:
            OPEN carts.txt
            FOR EACH line IN file:
                SPLIT line INTO [user, *items]
                IF items:
                    carts[user] = items
        EXCEPT FileNotFoundError:
            PASS
        RETURN carts

# main.py
@app.route('/cart')
FUNCTION cart():
    IF 'username' NOT IN session:
        REDIRECT TO login
    cart_items = cart_manager.get_cart(session['username'])
    products = product_manager.get_products()
    items = FILTER products WHERE str(p['id']) IN cart_items
    RENDER cart.html WITH items=items
```

**Key Libraries:**  
- `Flask` (session handling, template rendering).  
- File I/O for cart storage.

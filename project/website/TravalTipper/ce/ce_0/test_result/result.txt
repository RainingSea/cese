```pseudocode
FLASK_ROUTE '/travel_details'
    IF request method is POST THEN
        SET destination = request.form['destination']
        SET duration = request.form['duration']
        SET interests = request.form.getlist('interests')
        SET tips = CALL generate_tips(destination, interests) FROM TravelTipGenerator
        RENDER recommendations.html with tips
    ENDIF
    RENDER travel_details.html
END FLASK_ROUTE

CLASS TravelTipGenerator
    FUNCTION __init__()
        INITIALIZE tips list
        CALL load_tips()
    END FUNCTION

    FUNCTION generate_tips(destination: str, interests: list) -> list
        RETURN FILTER tips WHERE destination is in tip AND any interest in interests is in tip
    END FUNCTION

    FUNCTION load_tips() -> None
        TRY TO open 'travel_tips.txt' AS file
            LOAD tips FROM file
        CATCH FileNotFoundError
            SET tips = empty list
        END CATCH
    END FUNCTION
END CLASS
```

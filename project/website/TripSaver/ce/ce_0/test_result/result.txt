```python
CLASS UserManager:
    DEF login(username, password):
        TRY:
            OPEN users_file IN READ MODE AS f
            FOR EACH line IN f:
                SPLIT line INTO stored_user, stored_pass
                IF stored_user == username AND stored_pass == password:
                    RETURN True
        EXCEPT FileNotFoundError:
            RETURN False
        RETURN False

# Flask Route:
ROUTE /login (GET, POST):
    IF request.method == 'POST':
        GET username, password FROM request.form
        IF UserManager.login(username, password):
            SET session['username'] = username

    RENDER login.html
```
**Key Functionality:**

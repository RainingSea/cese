CLASS UserManager
    FUNCTION __init__()
        CALL load_users()

    FUNCTION load_users()
        IF users.txt exists
            OPEN users.txt
            FOR EACH line IN users.txt
                username, password = SPLIT line by '|'
                ADD username and password to users dictionary
            END FOR
        END IF

    FUNCTION register(username, password) -> bool
        IF username IS IN users dictionary
            RETURN False  // Username already exists
        ENDIF
        OPEN users.txt in append mode
        WRITE username + "|" + password to users.txt
        ADD username and password to users dictionary
        RETURN True  // Registration successful
END CLASS

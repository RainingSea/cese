        SET username TO request.form['username']
        SET password TO request.form['password']
        CALL user_manager.register(username, password)

        REDIRECT to the login page
    ENDIF
    RENDER registration.html
END FUNCTION

CLASS UserManager
    FUNCTION register(username, password)
        IF username IN self.users
            RETURN False
        ENDIF

        OPEN users_file IN append mode AS file
            WRITE username, ":", password TO file
        END OPEN

        ADD username TO self.users
        RETURN True
    END FUNCTION
END CLASS

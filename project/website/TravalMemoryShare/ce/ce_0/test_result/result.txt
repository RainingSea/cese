```plaintext
CLASS UserManager
    FUNCTION __init__()
        CALL load_users()

    FUNCTION load_users()
        IF users.txt does NOT exist
            RETURN empty list
        ENDIF
        OPEN users.txt AS file
            FOR each line IN file
                APPEND split(line, '|') TO users
            END FOR
        END OPEN file
        RETURN users

    FUNCTION login(username, password)
        FOR each user IN users
            IF user[0] == username AND user[1] == password
                RETURN True
            ENDIF
        END FOR
        RETURN False

CLASS AlbumManager
    FUNCTION __init__()
        CALL load_albums()

    FUNCTION load_albums()
        IF albums.txt does NOT exist
            RETURN empty list
        ENDIF
        OPEN albums.txt AS file
            FOR each line IN file
                APPEND split(line, '|') TO albums
            END FOR
        END OPEN file
        RETURN albums

    FUNCTION explore_albums()
        RETURN albums

ROUTE album_exploration
    albums = CALL album_manager.explore_albums()
    RENDER album_exploration.html WITH albums
```

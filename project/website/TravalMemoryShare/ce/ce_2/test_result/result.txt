FUNCTION User
    INITIALIZE username: STRING, password: STRING
    FUNCTION load_users() RETURNS LIST
        TRY
            OPEN 'users.txt'
            FOR EACH line IN file
                SPLIT line INTO username, password
                APPEND User(username, password) TO users
            END FOR
        EXCEPTION FileNotFoundError
            PASS
        RETURN users
    END FUNCTION

    FUNCTION save()
        OPEN 'users.txt' in append mode
            WRITE username AND password in 'username|password' format
        CLOSE file
    END FUNCTION
END FUNCTION

FUNCTION UserController
    FUNCTION register_user(username: STRING, password: STRING) RETURNS BOOLEAN
        users = User.load_users()
        IF ANY user.username EQUALS username IN users THEN
            RETURN False  # Username already exists
        END IF
        new_user = User(username, password)
        new_user.save()
        RETURN True  # Registration successful
    END FUNCTION
END FUNCTION

FUNCTION registration()
    IF request.method EQUALS 'POST' THEN
        username = request.form['username']
        password = request.form['password']
        user_controller = UserController()
        IF user_controller.register_user(username, password) THEN
            REDIRECT to login page
        ELSE
            RENDER registration.html with error message
        END IF
    END IF
    RENDER registration.html
END FUNCTION

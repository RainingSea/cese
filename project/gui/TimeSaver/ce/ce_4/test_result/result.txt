    FUNCTION __init__() -> NONE:
        self.shopping_lists = DataStorage.load_shopping_lists('shopping_lists.txt')
        self.items = DataStorage.load_items('items.txt')

    FUNCTION create_list(name: STRING) -> NONE:
        IF name NOT IN self.shopping_lists THEN
            APPEND name to self.shopping_lists
            self.save_data()

    FUNCTION delete_list(name: STRING) -> NONE:
        IF name IN self.shopping_lists THEN
            REMOVE name from self.shopping_lists
            IF name IN self.items THEN
                DELETE name from self.items
            self.save_data()

    FUNCTION save_data() -> NONE:
        DataStorage.save_shopping_lists('shopping_lists.txt', self.shopping_lists)

```plaintext
CLASS ReceiptManager
    INIT()
        CREATE empty list receipts
        CALL load_receipts()

    FUNCTION add_receipt(date: STRING, merchant: STRING, total: FLOAT)
        CREATE receipt as formatted string "date,merchant,total"
        APPEND receipt to receipts
        CALL save_receipts()

    FUNCTION search_receipts(query: STRING) -> LIST
        RETURN filtered list of receipts WHERE query is in receipt

    FUNCTION load_receipts()
        IF receipts.txt exists
            OPEN receipts.txt in read mode
            FOR each line in file
                APPEND trimmed line to receipts
            END FOR
            CLOSE file

    FUNCTION save_receipts()
        OPEN receipts.txt in write mode
        FOR each receipt in receipts
            WRITE receipt with newline to file
        END FOR
        CLOSE file
END CLASS
```

**Functionality of `Main` Class (for user interface interaction)**

```plaintext
CLASS Main
    INIT(master)
        SET master window title to "Receipt Manager"
        INSTANTIATE ReceiptManager as receipt_manager

        CREATE Label for "Date (YYYY-MM-DD)"
        CREATE Entry for DATE input

        CREATE Label for "Merchant"
        CREATE Entry for MERCHANT input

        CREATE Label for "Total Amount"
        CREATE Entry for TOTAL input

        CREATE Button "Save Receipt" with command linking to save_receipt()
        CREATE Label for "Search Receipts"
        CREATE Entry for SEARCH query input
        
        CREATE Button "Search" with command linking to search_receipts()

        CREATE Label for "Results"
        CREATE Text area for displaying results

    FUNCTION save_receipt() -> None
        GET date, merchant, total from respective Entries
        TRY
            CONVERT total to FLOAT
            CALL receipt_manager.add_receipt(date, merchant, total)
            SHOW SUCCESS message for saved receipt
            CALL clear_entries()
        EXCEPT ValueError
            SHOW ERROR message for invalid total

    FUNCTION search_receipts() -> None
        GET query from search Entry
        CALL receipt_manager.search_receipts(query)
        DISPLAY results in results_display area
END CLASS
```

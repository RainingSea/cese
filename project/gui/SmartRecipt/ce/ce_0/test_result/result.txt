```pseudocode
IMPORT tkinter as tk
IMPORT messagebox from tkinter
IMPORT os

CLASS Receipt:
    FUNCTION __init__(date: str, merchant: str, total_amount: float):
        SET self.date = date
        SET self.merchant = merchant
        SET self.total_amount = total_amount

CLASS Main:
    FUNCTION __init__(root):
        SET self.root = root
        SET self.root.title("Receipt Manager")
        SET self.receipts = self.load_receipts()      # Load receipts from file
        CALL self.create_widgets()                     # Create GUI widgets

    FUNCTION create_widgets():
        # Create GUI elements
        ADD label for "Date"
        CREATE date_entry input field
        ADD label for "Merchant Name"
        CREATE merchant_entry input field
        ADD label for "Total Amount"
        CREATE amount_entry input field
        
        # Create Save button
        CREATE save_button button and assign it to call save_receipt()

    FUNCTION save_receipt():
        SET date = GET value from date_entry
        SET merchant = GET value from merchant_entry
        SET total_amount = CONVERT value from amount_entry to float
        
        CALL self.add_receipt(date, merchant, total_amount)  # Add the receipt to list
        CALL messagebox.showinfo("Success", "Receipt saved successfully!")  # Show success message

    FUNCTION add_receipt(date: str, merchant: str, total_amount: float):
        CREATE a Receipt instance with provided details
        APPEND the new receipt to self.receipts
        CALL self.save_receipts_to_file()  # Persist data to file

    FUNCTION load_receipts():
        IF file 'receipts.txt' exists:
            OPEN 'receipts.txt' for reading
            FOR each line IN file:
                SPLIT line into components
                CREATE and RETURN a list of Receipt objects
        ELSE:
            RETURN an empty list

    FUNCTION save_receipts_to_file():
        OPEN 'receipts.txt' for writing
        FOR each receipt in self.receipts:
            WRITE formatted receipt data to file
```

- **messagebox:** A specific feature from tkinter for showing informational messages or prompts to the user.
- **os:** A standard library used to interact with the operating system, including file existence checks.

```pseudocode
    FUNCTION assign_task(task_id: INTEGER, user_id: INTEGER) -> NONE:
        IF task_id EXISTS IN tasks AND user_id EXISTS IN users THEN
            tasks[task_id].assignee = users[user_id].name
            CALL save_tasks()

    FUNCTION save_tasks() -> NONE:
        OPEN "tasks.txt" FOR WRITING AS file:
            FOR EACH task IN tasks:
                WRITE task.title + "|" + task.description + "|" + task.assignee + "|" + task.deadline + "|" + task.priority + "|" + task.status TO file
```

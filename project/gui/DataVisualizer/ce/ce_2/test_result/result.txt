        INITIALIZE data AS DataFrame()

    FUNCTION import_data(file_path: STRING):
        CALL data.load_data(file_path)

    FUNCTION create_visualization(type: STRING):
        data = CALL data.get_data()
        IF type EQUALS "Bar Chart":
            CREATE Bar Chart USING plt.bar(range(len(data)), data)
        ELSE IF type EQUALS "Line Graph":
            CREATE Line Graph USING plt.plot(data)
        ELSE IF type EQUALS "Scatter Plot":
            CREATE Scatter Plot USING plt.scatter(range(len(data)), data)
        ELSE IF type EQUALS "Pie Chart":

        CALL plt.show()

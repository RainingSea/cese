- **Pseudocode**:
  ```plaintext
  CLASS Shape
    METHOD __init__(shape_type: string, properties: dict)
      SET self.shape_type = shape_type
      SET self.properties = properties

    METHOD draw(canvas: Canvas)
      IF self.shape_type == 'rectangle'
        CALL canvas.create_rectangle(self.properties)
      ELSE IF self.shape_type == 'circle'
        EXTRACT x, y, radius FROM self.properties
        CALL canvas.create_oval(x - radius, y - radius, x + radius, y + radius, self.properties)
      ELSE IF self.shape_type == 'triangle'
        EXTRACT points FROM self.properties
        CALL canvas.create_polygon(points, self.properties)
      ELSE IF self.shape_type == 'polygon'
        EXTRACT points FROM self.properties
        CALL canvas.create_polygon(points, self.properties)

  CLASS ShapeManager
    METHOD __init__()
      SET self.shapes = []

    METHOD add_shape(shape: Shape)
      APPEND shape TO self.shapes
  ```

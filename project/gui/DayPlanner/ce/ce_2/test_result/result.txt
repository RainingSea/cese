        SET self.tasks = []

    FUNCTION add_task(description: str, priority: int, category: str, time_slot: str):
        CREATE new_task = Task(description, priority, category, time_slot)
        APPEND new_task to self.tasks
        CALL self.save_tasks()

    FUNCTION get_tasks():
        RETURN self.tasks

    FUNCTION save_tasks():
        OPEN 'tasks.txt' for writing AS file:
            FOR EACH task IN self.tasks:
                WRITE f"{task.description}|{task.priority}|{task.category}|{task.time_slot}\n" to file

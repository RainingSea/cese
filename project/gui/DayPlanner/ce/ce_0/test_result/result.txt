**Task Manager Class (task_manager.py)**
```plaintext
CLASS Task
    FUNCTION __init__(title: str, priority: str, category: str, time_slot: str)
        SET self.title = title
        SET self.priority = priority
        SET self.category = category
        SET self.time_slot = time_slot
    END FUNCTION
END CLASS

CLASS TaskManager
    FUNCTION __init__()
        INITIALIZE self.tasks as EMPTY LIST
    END FUNCTION

    FUNCTION add_task(task: Task)
        APPEND task to self.tasks
        CALL self.save_tasks()
    END FUNCTION

    FUNCTION load_tasks()
        IF "tasks.txt" EXISTS
            OPEN "tasks.txt" AS file
                FOR EACH line IN file
                    SPLIT line by "|"
                    CREATE task from title, priority, category, time_slot
                    APPEND task to self.tasks
                END FOR
            CLOSE file
        ENDIF
    END FUNCTION

    FUNCTION save_tasks()
        OPEN "tasks.txt" in WRITE mode
            FOR EACH task IN self.tasks
                WRITE f"{task.title}|{task.priority}|{task.category}|{task.time_slot}\n"
            END FOR
        CLOSE file
    END FUNCTION

    FUNCTION get_tasks()
        RETURN self.tasks
    END FUNCTION
END CLASS
```

**Main Application (main.py)**
```plaintext
CLASS Main
    FUNCTION __init__(root)
        SET self.root = root
        CALL self.task_manager = TaskManager()
        CALL self.create_widgets()
        CALL self.load_data()
    END FUNCTION

    FUNCTION create_widgets()
        CREATE Entry for task_title
        CREATE Dropdown for priority
        CREATE Dropdown for category
        CREATE Entry for time_slot
        CREATE Button for add_task
        CREATE Listbox for task_list
    END FUNCTION

    FUNCTION load_data()
        CALL self.task_manager.load_tasks()
    END FUNCTION

    FUNCTION add_task()
        SET title = self.task_title.get()
        SET priority = self.priority_var.get()
        SET category = self.category_var.get()
        SET time_slot = self.time_slot.get()
        IF all fields are filled
            CREATE new task
            CALL self.task_manager.add_task(new task)
            CALL self.update_task_list()
            CALL self.clear_inputs()
        ELSE
            SHOW warning dialog
    END FUNCTION

    FUNCTION update_task_list()
        CLEAR task_list
        FOR EACH task IN self.task_manager.get_tasks()
            ADD task string to task_list
        END FOR
    END FUNCTION
END CLASS
```

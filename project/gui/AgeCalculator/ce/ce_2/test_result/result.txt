    @staticmethod
    def get_days_in_month(month: int, year: int) -> int:
        # Calculate the number of days in a month accounting for leap years
        if month == 2:
            return 29 if (year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)) else 28
        elif month in [4, 6, 9, 11]:
            return 30
        else:
            return 31

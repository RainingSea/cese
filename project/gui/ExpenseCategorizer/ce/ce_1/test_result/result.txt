CLASS ExpenseManager:
    FUNCTION __init__():
        self.expenses = []
        self.categories = []
        CALL self.load_expenses()
        CALL self.load_categories()

    FUNCTION add_expense(amount: float, description: str, category: str) -> None:
        APPEND (amount, description, category) to self.expenses
        CALL self.save_expenses()

    FUNCTION load_expenses() -> None:
        TRY:
            OPEN 'expenses.txt' AS file:
                FOR EACH line IN file:
                    SPLIT line by '|'
                    ASSIGN amount, description, category
                    APPEND (float(amount), description, category) to self.expenses
        EXCEPT FileNotFoundError:
            SET self.expenses to []

    FUNCTION load_categories() -> None:
        TRY:
            OPEN 'categories.txt' AS file:
                SET self.categories to [line.strip() FOR line IN file]
        EXCEPT FileNotFoundError:
            SET self.categories to []

    FUNCTION save_expenses() -> None:
        OPEN 'expenses.txt' IN write mode AS file:
            FOR EACH expense IN self.expenses:
                WRITE expense formatted to file

    FUNCTION get_summary() -> dict:
        INIT summary as an empty dictionary
        FOR EACH amount, description, category IN self.expenses:
            IF category is IN summary:
                INCREMENT summary[category] by amount
            ELSE:
                SET summary[category] to amount
        RETURN summary

CLASS Main:
    FUNCTION __init__(master):
        self.master = master
        SET WINDOW TITLE to "Expense Categorizer"
        CREATE instance of ExpenseManager
        CALL self.create_widgets()

    FUNCTION create_widgets():
        INIT Widgets for amount, description, category_combobox, and buttons
        CALL self.load_expenses_to_listbox()

    FUNCTION load_expenses_to_listbox():
        CLEAR self.expense_listbox
        FOR EACH amount, description, category IN self.expense_manager.expenses:
            INSERT formatted string into self.expense_listbox

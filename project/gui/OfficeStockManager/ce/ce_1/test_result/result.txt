     ```pseudocode
     CLASS InventoryManager
         FUNCTION __init__(file_path: str)
             SET self.file_path = file_path
             SET self.inventory = load_inventory()

         FUNCTION add_item(item_name: str, item_type: str, quantity: int)
             APPEND {'item_name': item_name, 'item_type': item_type, 'quantity': quantity} TO self.inventory
             CALL save_inventory()

         FUNCTION search_item(item_name: str) -> dict
             FOR item IN self.inventory
                 IF item['item_name'] == item_name
                     RETURN item
             RETURN {}
             
         FUNCTION load_inventory() -> list
             IF NOT os.path.exists(self.file_path)
                 RETURN []
             OPEN self.file_path AS file
                 SET items = []
                 FOR line IN file
                     SPLIT line BY ','
                     APPEND {'item_name': item_name, 'item_type': item_type, 'quantity': INT(quantity)} TO items
                 RETURN items
             
         FUNCTION save_inventory()
             OPEN self.file_path AS file FOR writing
                 FOR item IN self.inventory
                     WRITE f"{item['item_name']},{item['item_type']},{item['quantity']}\n" TO file
     ```

```
CLASS InventoryManager
    METHOD __init__(filename: str)
        SET self.filename = filename
        SET self.items = []
        CALL load_items()

    METHOD add_item(name: str, category: str, quantity: int)
        SET item = {'name': name, 'category': category, 'quantity': quantity}
        APPEND item TO self.items
        CALL save_items()

    METHOD search_item(name: str) -> Dict[str, Any]
        FOR item IN self.items
            IF item['name'] == name
                RETURN item
        RETURN {}

    METHOD load_items()
        TRY
            OPEN file as self.filename in read mode
            SET self.items = json.load(file)
        EXCEPT FileNotFoundError
            SET self.items = []

    METHOD save_items()
        OPEN file as self.filename in write mode
        json.dump(self.items, file, indent=4)
```

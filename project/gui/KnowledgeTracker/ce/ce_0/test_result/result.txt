    FUNCTION __init__(file_path):
        SET self.file_path = file_path
        SET self.knowledge_data = load_knowledge()

    FUNCTION load_knowledge() RETURNS list:
        IF os.path.exists(self.file_path):
            OPEN self.file_path AS file:
                RETURN json.load(file)
        RETURN []

    FUNCTION add_knowledge(entry):
        APPEND entry to self.knowledge_data
        CALL save_knowledge()

    FUNCTION retrieve_knowledge() RETURNS list:
        RETURN self.knowledge_data

    FUNCTION save_knowledge():
        OPEN self.file_path FOR writing AS file:
            json.dump(self.knowledge_data, file, indent=4)

MAIN FUNCTION (execution starts):

    CALL app.main() to start GUI

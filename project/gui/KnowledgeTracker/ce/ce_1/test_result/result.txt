```plaintext
CLASS DataStorage
    FUNCTION __init__()
        DEFINE file_map AS DICTIONARY
            SET file_map['theories'] TO 'theories.txt'
            SET file_map['concepts'] TO 'concepts.txt'
            SET file_map['experiments'] TO 'experiments.txt'
    
    FUNCTION load_data(type: STRING)
        IF type NOT IN file_map
            RAISE ValueError("Invalid type specified.")
        ENDIF
        SET file_path AS file_map[type]
        IF NOT os.path.exists(file_path)
            RETURN EMPTY LIST
        ENDIF
        READ file_path INTO file
        RETURN EACH line.strip() FROM file.readlines()

    FUNCTION save_data(type: STRING, knowledge: STRING)
        IF type NOT IN file_map
            RAISE ValueError("Invalid type specified.")
        ENDIF
        SET file_path AS file_map[type]
        OPEN file_path IN APPEND MODE
            WRITE knowledge + '\n'
        END OPEN

CLASS KnowledgeManager
    FUNCTION __init__()
        INITIALIZE data_storage AS DataStorage()
        SET theories TO data_storage.load_data('theories')
        SET concepts TO data_storage.load_data('concepts')
        SET experiments TO data_storage.load_data('experiments')

    FUNCTION add_knowledge(type: STRING, knowledge: STRING)
        IF type EQUALS 'theories'
            APPEND knowledge TO theories
        ELSE IF type EQUALS 'concepts'
            APPEND knowledge TO concepts
        ELSE IF type EQUALS 'experiments'
            APPEND knowledge TO experiments
        ELSE
            RAISE ValueError("Invalid type specified.")
        ENDIF
        CALL data_storage.save_data(type, knowledge)

    FUNCTION retrieve_knowledge(type: STRING)
        IF type EQUALS 'theories'
            RETURN theories
        ELSE IF type EQUALS 'concepts'
            RETURN concepts
        ELSE IF type EQUALS 'experiments'
            RETURN experiments
        ELSE
            RAISE ValueError("Invalid type specified.")
        ENDIF
```

- `DataStorage`: Handles data saving and loading to/from text files.
- `KnowledgeManager`: Manages knowledge across different categories (theories, concepts, experiments).

```plaintext
CLASS KnowledgeManager
    FUNCTION update_knowledge(type: STRING, old_knowledge: STRING, new_knowledge: STRING)
        IF type EQUALS 'theories'
            SET index AS theories.index(old_knowledge)
            SET theories[index] TO new_knowledge
        ELSE IF type EQUALS 'concepts'
            SET index AS concepts.index(old_knowledge)
            SET concepts[index] TO new_knowledge
        ELSE IF type EQUALS 'experiments'
            SET index AS experiments.index(old_knowledge)
            SET experiments[index] TO new_knowledge
        ELSE
            RAISE ValueError("Invalid type specified.")
        ENDIF
        CALL save_all()

    FUNCTION save_all()
        FOR EACH theory IN theories
            CALL data_storage.save_data('theories', theory)
        FOR EACH concept IN concepts
            CALL data_storage.save_data('concepts', concept)
        FOR EACH experiment IN experiments
            CALL data_storage.save_data('experiments', experiment)
```

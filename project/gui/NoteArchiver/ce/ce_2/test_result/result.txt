```
CLASS NotebookManager
    FUNCTION __init__()
        CALL load_notebooks()
        CALL load_archived_notes()

    FUNCTION load_notebooks()
        IF notebooks.txt EXISTS THEN
            OPEN notebooks.txt AS file
                READ lines INTO notebooks LIST
                RETURN notebooks LIST
        ELSE
            RETURN EMPTY LIST

    FUNCTION load_archived_notes()
        IF archived_notes.txt EXISTS THEN
            OPEN archived_notes.txt AS file
                LOAD DATA INTO archived_notes DICTIONARY
                RETURN archived_notes DICTIONARY
        ELSE
            RETURN EMPTY DICTIONARY

    FUNCTION archive_notebook(notebook_name)
        IF notebook_name IN notebooks THEN
            REMOVE notebook_name FROM notebooks
            archived_notes[notebook_name] = []   // Initializes as empty archive
            CALL save_archived_notes()

    FUNCTION save_archived_notes()
        OPEN archived_notes.txt FOR WRITING AS file
            DUMP archived_notes INTO file USING json

FUNCTION main()
    CREATE ROOT WINDOW USING Tk()
    CREATE INSTANCE OF Main WITH root
    CALL root.mainloop()

CLASS Main
    FUNCTION __init__(master)
        CREATE UI components USING tkinter (Listbox, Button, Entry)
        CALL load_notebooks()

    FUNCTION load_notebooks()
        // Load notebooks from file into Listbox
        CALL load_notebooks FROM NotebookManager

    ...
```

- `NotebookManager` handles the loading, archiving, and saving of notebook data.
- Utilizes the **tkinter** library to create a graphical user interface (GUI).
- Implements **JSON** for data persistence in files (`archived_notes.txt` and `notebooks.txt`).

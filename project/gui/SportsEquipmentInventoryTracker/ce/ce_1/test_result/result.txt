CLASS InventoryManager
    FUNCTION __init__()
        CALL load_equipment()
        CALL load_alerts()
    END FUNCTION

    FUNCTION load_equipment() -> list
        TRY
            OPEN 'equipment_data.json' AS file FOR READING
                SET data = JSON.PARSE(file)
                RETURN [CREATE Equipment from item FOR EACH item IN data]
            END OPEN
        EXCEPTION FileNotFoundError
            RETURN []  // Return empty list if file not found
    END FUNCTION

    FUNCTION load_alerts() -> list
        TRY
            OPEN 'alerts.json' AS file FOR READING
                SET data = JSON.PARSE(file)
                RETURN [CREATE Alert from item FOR EACH item IN data]
            END OPEN
        EXCEPTION FileNotFoundError
            RETURN []  // Return empty list if file not found
    END FUNCTION

    FUNCTION add_equipment(name: str, quantity: int, condition: str, location: str)
        CREATE new_equipment = Equipment(name, quantity, condition, location)
        APPEND new_equipment TO self.equipment
        CALL save_equipment()
    END FUNCTION

    FUNCTION update_equipment(name: str, quantity: int, condition: str, location: str)
        FOR EACH item IN self.equipment
            IF item.name == name
                SET item.quantity = quantity
                SET item.condition = condition
                SET item.location = location
                CALL save_equipment()
                RETURN
            END IF
        END FOR
        CALL messagebox.showerror("Error", "Equipment not found.")
    END FUNCTION

    FUNCTION get_equipment() -> list
        RETURN self.equipment
    END FUNCTION

    FUNCTION set_alert(name: str, alert_type: str)
        CREATE new_alert = Alert(name, alert_type)
        APPEND new_alert TO self.alerts
        CALL save_alerts()
    END FUNCTION

    FUNCTION get_alerts() -> list
        RETURN self.alerts
    END FUNCTION

    FUNCTION save_equipment()
        OPEN 'equipment_data.json' FOR WRITING AS file
            WRITE JSON.DUMP([GET VARS(item) FOR EACH item IN self.equipment], file)
        END OPEN
    END FUNCTION

    FUNCTION save_alerts()
        OPEN 'alerts.json' FOR WRITING AS file
            WRITE JSON.DUMP([GET VARS(item) FOR EACH item IN self.alerts], file)
        END OPEN
    END FUNCTION

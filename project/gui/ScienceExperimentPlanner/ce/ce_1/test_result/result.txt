**Class: ExperimentManager**

```
CLASS ExperimentManager
    FUNCTION __init__()
        SET self.experiments = EMPTY LIST
    END FUNCTION
    
    FUNCTION add_experiment(experiment)
        APPEND experiment TO self.experiments
        CALL self.save_experiments()
    END FUNCTION

    FUNCTION edit_experiment(experiment_id, updated_experiment)
        IF experiment_id IS BETWEEN 0 AND LENGTH(self.experiments) - 1
            SET self.experiments[experiment_id] = updated_experiment
            CALL self.save_experiments()
        END IF
    END FUNCTION

    FUNCTION load_experiments()
        IF FILE "experiments.txt" EXISTS
            OPEN FILE "experiments.txt" IN READ MODE AS file
                FOR EACH line IN file
                    SPLIT line BY "|"
                    SET title, objectives, materials, procedures, status AS SPLIT RESULT
                    SET materials_list = SPLIT materials BY ","
                    CREATE Experiment OBJECT experiment(title, objectives, materials_list, procedures, status)
                    APPEND experiment TO self.experiments
                END FOR
            CLOSE FILE
        END IF
    END FUNCTION

    FUNCTION save_experiments()
        OPEN FILE "experiments.txt" IN WRITE MODE AS file
            FOR EACH experiment IN self.experiments
                SET materials_str = JOIN experiment.materials BY ","
                WRITE TO file f"{experiment.title}|{experiment.objectives}|{materials_str}|{experiment.procedures}|{experiment.status}\n"
            END FOR
        CLOSE FILE
    END FUNCTION
END CLASS
```

```
CLASS Main
    FUNCTION __init__(root)
        SET self.root = root
        SET self.root.title("Science Experiment Planner")
        CREATE ExperimentManager OBJECT experiment_manager
        CALL self.experiment_manager.load_experiments()
        CALL self.create_widgets()
    END FUNCTION

    FUNCTION create_widgets()
        CREATE Listbox widget self.experiment_listbox
        CALL self.experiment_listbox.pack(fill=tk.BOTH, expand=True)
        
        CREATE Button widget self.add_button WITH label "Add Experiment" AND command self.add_experiment
        CALL self.add_button.pack(side=tk.LEFT)

        CREATE Button widget self.edit_button WITH label "Edit Experiment" AND command self.edit_experiment
        CALL self.edit_button.pack(side=tk.LEFT)

        CREATE Button widget self.delete_button WITH label "Delete Experiment" AND command self.delete_experiment
        CALL self.delete_button.pack(side=tk.LEFT)

        CALL self.update_experiment_list()
    END FUNCTION

    FUNCTION update_experiment_list()
        CALL self.experiment_listbox.delete(0, tk.END)
        FOR EACH experiment IN self.experiment_manager.experiments
            CALL self.experiment_listbox.insert(tk.END, experiment.title)
        END FOR
    END FUNCTION
END CLASS
```

CLASS Experiment:
    FUNCTION __init__(id, objective, materials, procedure, expected_results, status="Pending"):
        SET self.id = id
        SET self.objective = objective
        SET self.materials = materials
        SET self.procedure = procedure
        SET self.expected_results = expected_results
        SET self.status = status

    FUNCTION to_string():
        RETURN formatted string of id, objective, materials, procedure, expected_results, status


CLASS ExperimentManager:
    FUNCTION __init__():
        INITIALIZE self.experiments as empty list
        CALL load_experiments()

    FUNCTION add_experiment(objective, materials, procedure, expected_results):
        SET experiment_id = length of self.experiments + 1
        CREATE new_experiment as Experiment(experiment_id, objective, materials, procedure, expected_results)
        APPEND new_experiment to self.experiments
        CALL save_experiments()

    FUNCTION load_experiments():
        IF 'experiments.txt' exists:
            OPEN 'experiments.txt' as file
            FOR each line in file:
                SPLIT line by ',' into id, objective, materials, procedure, expected_results, status
                CREATE experiment as Experiment(int(id), objective, materials, procedure, expected_results, status)
                APPEND experiment to self.experiments

    FUNCTION save_experiments():
        OPEN 'experiments.txt' in write mode as file
        FOR each experiment in self.experiments:
            WRITE experiment.to_string() to file
```

CLASS Observation:

        SET self.experiment_id = experiment_id
        SET self.observation = observation
        SET self.timestamp = timestamp

    FUNCTION to_string():
        RETURN formatted string of experiment_id, observation, timestamp


CLASS ObservationManager:
    FUNCTION __init__():
        INITIALIZE self.observations as empty list
        CALL load_observations()

    FUNCTION record_observation(experiment_id, observation):
        GET current datetime as timestamp in iso format
        CREATE new_observation as Observation(experiment_id, observation, timestamp)
        APPEND new_observation to self.observations
        CALL save_observations()

    FUNCTION load_observations():
        IF 'observations.txt' exists:
            OPEN 'observations.txt' as file
            FOR each line in file:
                SPLIT line by ',' into experiment_id, observation, timestamp
                CREATE obs as Observation(int(experiment_id), observation, timestamp)
                APPEND obs to self.observations

    FUNCTION save_observations():
        OPEN 'observations.txt' in write mode as file
        FOR each observation in self.observations:
            WRITE observation.to_string() to file
```

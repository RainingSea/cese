    OPEN filename in write mode AS file:
        FOR each key-value pair IN experiment_data:
            WRITE key and value to file in format 'key|value'

FUNCTION load_from_file(filename: str) -> dict:
    INITIALIZE experiment_data as empty dictionary
    IF filename exists:
        OPEN filename in read mode AS file:
            FOR each line IN file:
                SPLIT line by '|' INTO key, value
                SET experiment_data[key] = value
    RETURN experiment_data

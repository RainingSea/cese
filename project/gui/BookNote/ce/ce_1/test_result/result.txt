     CLASS BookManager

         END FUNCTION

         FUNCTION add_book(title: string, author: string, publication_date: string)
             CREATE new_book from Book with title, author, publication_date
             APPEND new_book to self.books
             CALL save_books()
         END FUNCTION

         FUNCTION load_books()
             TRY TO open 'books.txt' for reading
                 FOR EACH line in file
                     SPLIT line by '|'
                     CREATE Book with title, author, publication_date
                     APPEND the new Book to self.books
                 END FOR
             CATCH FileNotFoundError
                 DO NOTHING
             END TRY
         END FUNCTION

         FUNCTION save_books()
             OPEN 'books.txt' for writing
             FOR EACH book in self.books
                 WRITE formatted book.title, book.author, book.publication_date to file
             END FOR
         END FUNCTION
     END CLASS

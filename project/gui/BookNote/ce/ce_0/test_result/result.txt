CLASS BookManager
    FUNCTION __init__()
        SET self.books = self.load_books()
    
    FUNCTION add_book(title: STRING, author: STRING, pub_date: STRING)
        OPEN "books.txt" FOR APPEND AS file
            WRITE title + "|" + author + "|" + pub_date + "\n"
        CLOSE file
        ADD (title, author, pub_date) TO self.books

    FUNCTION load_books() -> LIST
        IF "books.txt" NOT EXISTS
            RETURN []
        OPEN "books.txt" FOR READ AS file
            FOR EACH line IN file
                SPLIT line BY "|" INTO title, author, pub_date
                ADD (title, author, pub_date) TO list
            RETURN list

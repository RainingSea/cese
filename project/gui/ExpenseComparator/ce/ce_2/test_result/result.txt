CLASS ExpenseManager:

        SET self.expenses TO empty list

    FUNCTION add_expense(date: String, amount: Float, category: String) -> None
        CREATE expense OBJECT of class Expense(date, amount, category)
        APPEND expense TO self.expenses

    FUNCTION get_expenses(start_date: String, end_date: String) -> List[Expense]
        FOR each expense IN self.expenses
            IF start_date <= expense.date <= end_date THEN
                ADD expense TO result_list
        RETURN result_list

FUNCTION load_expenses() -> List[dict]
    TRY
        OPEN 'expenses.txt' FOR reading
        SET expenses TO empty list
        FOR each line IN file
            SPLIT line by '|' INTO date, amount, category
            ADD {'date': date, 'amount': Float(amount), 'category': category} TO expenses
        RETURN expenses
    EXCEPT FileNotFoundError
        RETURN empty list

FUNCTION load_categories() -> List[str]
    TRY
        OPEN 'categories.txt' FOR reading
        RETURN [line.strip() FOR line IN file]
    EXCEPT FileNotFoundError
        RETURN empty list

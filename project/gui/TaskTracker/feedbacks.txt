4. |Case|:**add_task**
```pseudocode
CLASS Task
    FUNCTION __init__(id: int, title: str, description: str, deadline: str, priority: str, category: str)
        ASSIGN task ID
        ASSIGN task title
        ASSIGN task description
        ASSIGN task deadline
        ASSIGN task priority
        ASSIGN task category
    END FUNCTION
END CLASS

CLASS TaskManager
    FUNCTION __init__()
        INITIALIZE tasks as empty list
        CALL load_tasks() to load saved tasks from file
    END FUNCTION

    FUNCTION add_task(title: str, description: str, deadline: str, priority: str, category: str)
        CALCULATE task_id as length of tasks + 1
        CREATE new_task as instance of Task using given parameters
        APPEND new_task to tasks
        CALL save_tasks()
    END FUNCTION

    FUNCTION edit_task(task_id: int, title: str, description: str, deadline: str, priority: str, category: str)
        FOR EACH task IN tasks
            IF task.id equals task_id
                UPDATE task.title
                UPDATE task.description
                UPDATE task.deadline
                UPDATE task.priority
                UPDATE task.category
                CALL save_tasks()
                RETURN
            ENDIF
        ENDFOR
    END FUNCTION

    FUNCTION load_tasks()
        IF tasks.json file exists
            OPEN tasks.json AS file
                LOAD tasks_data from the file
                FOR EACH task IN tasks_data
                    CREATE Task using task data and APPEND to tasks
                ENDFOR
            CLOSE file
        ENDIF
    END FUNCTION

    FUNCTION save_tasks()
        OPEN tasks.json AS file
            DUMP tasks to the file in JSON format
        CLOSE file
    END FUNCTION
END CLASS
```

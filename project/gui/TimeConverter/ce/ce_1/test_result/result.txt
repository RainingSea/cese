```plaintext
CLASS TimeConverter:
    FUNCTION convert_time(source_time: str, source_tz: str, target_tz: str, format: str) -> str:
        INITIALIZE source_timezone with pytz.timezone(source_tz)
        INITIALIZE target_timezone with pytz.timezone(target_tz)

        PARSE naive_time from source_time using "%H:%M" if format is 24-hour
        PARSE naive_time from source_time using "%I:%M %p" if format is 12-hour

        LOCALIZE naive_time to source_timezone
        CONVERT localized_time to target_time in target_timezone
        
        IF format is "12-hour":
            RETURN target_time formatted as "%I:%M %p"
        ELSE:
            RETURN target_time formatted as "%H:%M"

    FUNCTION save_conversion(source_time: str, source_tz: str, target_tz: str, converted_time: str, format: str):
        OPEN conversion_history.txt in append mode
        WRITE source_time, source_tz, target_tz, converted_time, format to file

    FUNCTION load_history() -> list:
        TRY to OPEN conversion_history.txt in read mode
            RETURN lines from file
        EXCEPT FileNotFoundError:
            RETURN empty list

    FUNCTION clear_history():
        OPEN conversion_history.txt in write mode
        CLOSE file (this clears the content)

    FUNCTION get_timezones() -> list:
        RETURN all timezones from pytz.all_timezones
```

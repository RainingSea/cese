```plaintext
        CONFIGURE frame for password input

    END FUNCTION

    FUNCTION add_vault()
        PROMPT user for vault name (simpledialog)
        IF vault_name IS NOT EMPTY
            CALL vault_manager.create_vault(vault_name)  // Create a new vault
            CALL update_vault_list()                      // Refresh list of vaults in UI
        ENDIF
    END FUNCTION

    FUNCTION add_password()
        FETCH selected vault from Listbox
        READ password from Entry field
        IF vault_name AND password ARE NOT EMPTY
            CALL vault_manager.add_password(vault_name, password)  // Store password associated with vault
            CLEAR password entry field
        ENDIF
    END FUNCTION

    FUNCTION update_vault_list()
        CLEAR Listbox
        FOR each vault IN vault_manager.vaults
            ADD vault name to Listbox
        END FOR
    END FUNCTION
END CLASS

CLASS VaultManager
    FUNCTION __init__()
        INITIALIZE empty list for vaults
    END FUNCTION

    FUNCTION load_vaults()
       IF "vaults.txt" EXISTS
            OPEN "vaults.txt" AS file
            READ all lines and TRIM whitespace
            ASSIGN lines to vaults list
       ENDIF
    END FUNCTION

    FUNCTION create_vault(name: str) -> None
        IF name NOT IN vaults
            APPEND name to vaults
            OPEN "vaults.txt" in append mode
            WRITE name to file
            OPEN corresponding vault file (name.txt) in write mode
            WRITE empty contents for the new vault file
        ENDIF
    END FUNCTION

    FUNCTION add_password(vault_name: str, password: str) -> None
        CALL PasswordEncryption.encrypt(password) // Encrypt the password
        OPEN corresponding vault file in append mode
        WRITE encrypted password to file
    END FUNCTION
END CLASS

CLASS PasswordEncryption
    FUNCTION encrypt(password: str) -> str
        GENERATE encryption key using Fernet
        CREATE Fernet cipher object with key
        ENCRYPT password and RETURN as string
    END FUNCTION

    FUNCTION decrypt(encrypted_password: str) -> str
        RECREATE key (this part limits decryption to original encryption context)
        CREATE Fernet cipher object with key
        DECRYPT password and RETURN as string
    END FUNCTION
END CLASS
```

The passwords are encrypted using the `cryptography` library's Fernet class, which provides symmetric encryption, ensuring that only those who possess the key can access the passwords. The user interface is created using Tkinter for ease of interaction.

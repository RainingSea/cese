```
CLASS DataStorage:
    FUNCTION save_scores(score: INTEGER) -> NONE:
        OPEN 'scores.txt' in append mode AS file:
            WRITE score followed by newline to file

    FUNCTION load_scores() -> LIST:
        IF 'scores.txt' does NOT exist:
            RETURN empty list
        OPEN 'scores.txt' in read mode AS file:
            RETURN list of integers parsed from each line in file

    FUNCTION load_vehicles() -> LIST:
        INITIALIZE vehicles AS empty list
        IF 'vehicles.txt' does NOT exist:
            RETURN vehicles
        OPEN 'vehicles.txt' in read mode AS file:
            FOR each line in file:
                SPLIT line by '|' INTO name, handling, acceleration, top_speed
                APPEND Vehicle object CREATED with name, handling, acceleration, top_speed to vehicles 
        RETURN vehicles
```

CLASS GameBoard
    FUNCTION __init__()
        INITIALIZE tiles as empty list

    FUNCTION initialize_board()
        SET tiles to a 4x4 grid filled with zeros
        CALL generate_tile()  // Generate the first tile
        CALL generate_tile()  // Generate the second tile

    FUNCTION generate_tile()
        SET empty_tiles to list of coordinates where tiles are 0
        IF empty_tiles is not empty
            SELECT random (i, j) from empty_tiles
            SET tiles[i][j] to 4 with probability 0.1, else set to 2

CLASS Game
    FUNCTION __init__()
        INITIALIZE board as new GameBoard
        INITIALIZE score to 0

    FUNCTION start_game()
        CALL board.initialize_board()  // Initialize the game board

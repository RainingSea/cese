            self.start_time = 0.0
            self.elapsed_time = 0.0

        FUNCTION start():
            self.start_time = time.now()

        FUNCTION stop():
            self.elapsed_time = time.now() - self.start_time

        FUNCTION get_time():
            RETURN self.elapsed_time

    CLASS Game:
        FUNCTION start_game(image_path, difficulty):
            CALL self.puzzle.create_puzzle(image_path, difficulty)
            CALL self.timer.start()

        FUNCTION save_progress():
            progress_data = {
                'pieces': [ (piece.get_position(), piece.image) FOR piece IN self.puzzle.pieces ],
                'elapsed_time': self.timer.get_time()

```plaintext
        IF 'progress.txt' EXISTS
            OPEN 'progress.txt' FOR READING
                FOR EACH line IN file
                    SPLIT line BY '|'
                    ASSIGN key to first PART
                    ASSIGN value to second PART
                    IF key == 'score'
                        SET self.score TO INTEGER(value)
                    ELSE IF key == 'timer'
                        SET self.timer TO INTEGER(value)
                    ELSE IF key == 'difficulty'
                        SET self.difficulty TO value
                END FOR
            END OPEN
END CLASS
```

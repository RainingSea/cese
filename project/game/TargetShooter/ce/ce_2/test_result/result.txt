                    SPLIT line BY '|' INTO name AND score
                    APPEND new Score(name, INTEGER(score)) TO scores
        EXCEPT FileNotFoundError
            INITIALIZE scores AS EMPTY LIST

    FUNCTION save_score(score AS INTEGER)
        PROMPT user "Enter your name: "
        INITIALIZE new_score AS Score(user_input, score)

        OPEN 'scores.txt' FOR APPENDING AS f
            WRITE new_score.player_name + '|' + new_score.score_value TO f

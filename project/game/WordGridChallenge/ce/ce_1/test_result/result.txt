CLASS ScoreManager

                    ASSIGN name and score
                    ADD name and integer score to scores dictionary
        EXCEPT FileNotFoundError
            SET scores to empty dictionary

    FUNCTION save_score(player: str, score: int)
        ASSIGN player and score to scores dictionary
        OPEN 'scores.txt' FOR WRITING
            FOR each name and score in scores dictionary
                WRITE name and score in "name:score" format

    FUNCTION load_word_list()
        TRY
            OPEN 'word_list.txt' FOR READING
                RETURN list of stripped lines from file
        EXCEPT FileNotFoundError
            RETURN empty list

KEY FUNCTIONALITY:
- Manages scores and loads/saves them to 'scores.txt'.
